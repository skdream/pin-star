    61→export function DictationView() {
    62→  const [stage, setStage] = useState<Stage>("setup");
    63→  const [queue, setQueue] = useState<WordDTO[]>([]);
    64→  const [attempts, setAttempts] = useState<Attempt[]>([]);
    65→  const [starting, setStarting] = useState(false);
    66→
    67→  const setView = useAppStore((s) => s.setView);
    68→
    69→  const handleStart = useCallback(async (req: DictationWordsReq) => {
    70→    setStarting(true);
    71→    try {
    72→      const { words } = await postDictationWords(req);
    73→      if (!words?.length) {
    74→        toast("这个词单里还没有单词，换一个试试～");
    75→        return;
    76→      }
    77→      setQueue(words);
    78→      setAttempts([]);
    79→      setStage("writing");
    80→    } catch {
    81→      toast.error("获取听写词单失败，请稍后再试");
    82→    } finally {
    83→      setStarting(false);
    84→    }
    85→  }, []);
    86→
    87→  const handleFinish = useCallback((a: Attempt[]) => {
    88→    setAttempts(a);
    89→    setStage("result");
    90→  }, []);
    91→
    92→  const retryWrong = useCallback(() => {
    93→    const seen = new Set<string>();
    94→    const wrongWords: WordDTO[] = [];
    95→    for (const a of attempts) {
    96→      if (!a.correct && !seen.has(a.word.id)) {
    97→        seen.add(a.word.id);
    98→        wrongWords.push(a.word);
    99→      }
   100→    }
   101→    if (!wrongWords.length) return;
   102→    setQueue(wrongWords);
   103→    setStage("writing");
   104→  }, [attempts]);
   105→
   106→  const backToSetup = useCallback(() => {
   107→    setQueue([]);
   108→    setAttempts([]);
   109→    setStage("setup");
   110→  }, []);
   111→
   112→  if (stage === "writing") {
   113→    return (
   114→      <WritingStage
   115→        key={queue.map((w) => w.id).join(",")}
   116→        words={queue}
   117→        onFinish={handleFinish}
   118→        onQuit={backToSetup}
   119→      />
   120→    );
   121→  }
   122→
   123→  if (stage === "result") {
   124→    return (
   125→      <ResultStage
   126→        attempts={attempts}
   127→        onRetryWrong={retryWrong}
   128→        onChangeList={backToSetup}
   129→        onHome={() => setView("home")}
   130→      />
   131→    );
   132→  }
   133→
   134→  return <SetupStage starting={starting} onStart={handleStart} />;
   135→}

   472→function HintArea({ level, word }: { level: HintLevel; word: WordDTO }) {
   473→  if (level === "first") {
   474→    const masked = word.headword
   475→      .split("")
   476→      .map((c, i) => (i === 0 ? c : "_"))
   477→      .join(" ");
   478→    return <p className="font-mono text-2xl font-bold tracking-[0.25em] text-foreground/80">{masked}</p>;
   479→  }
   480→  if (level === "chinese") {
   481→    return <p className="text-lg font-medium text-foreground/80">释义：{word.translation}</p>;
   482→  }
   483→  if (level === "phonemes") {
   484→    return <PhonemeBlocks word={word} size="sm" hideLetters />;
   485→  }
   486→  return <p className="text-sm text-muted-foreground">仔细听，把它拼出来吧！</p>;
   487→}
   488→
   489→function WritingStage({
   490→  words,
   491→  onFinish,
   492→  onQuit,
   493→}: {
   494→  words: WordDTO[];
   495→  onFinish: (attempts: Attempt[]) => void;
   496→  onQuit: () => void;
   497→}) {
   498→  const { speak, speakTimes, stop, speaking } = useSpeech();
   499→
   500→  const settings = useAppStore((s) => s.user.settings);
   501→  const playCount = settings.playCount ?? 2;
   502→  const speed = settings.speed ?? "normal";
   503→  const hintLevel = settings.hintLevel ?? "none";
   504→  const autoNext = settings.autoNext ?? true;
   505→
   506→  const [index, setIndex] = useState(0);
   507→  const [answer, setAnswer] = useState("");
   508→  const [checking, setChecking] = useState(false);
   509→  const [result, setResult] = useState<CheckResult | null>(null);
   510→
   511→  const attemptsRef = useRef<Attempt[]>([]);
   512→  const startRef = useRef(Date.now());
   513→  const inputRef = useRef<HTMLInputElement>(null);
   514→
   515→  const word = words[index];
   516→  const progress = ((index + (result ? 1 : 0)) / Math.max(1, words.length)) * 100;
   517→
   518→  /* 换词时：重置计时 + 自动播放 N 遍 + 聚焦输入框 */
   519→  useEffect(() => {
   520→    if (!word) return;
   521→    startRef.current = Date.now();
   522→    if (!result) {
   523→      speakTimes(word.headword, { times: playCount, speed, intervalMs: 1500 });
   524→    }
   525→    const t = setTimeout(() => inputRef.current?.focus(), 150);
   526→    return () => clearTimeout(t);
   527→  }, [word, result, playCount, speed, speakTimes]);
   528→
   529→  const submit = useCallback(async () => {
   530→    const val = answer.trim();
   531→    if (!val) {
   532→      toast("先写一写再提交吧～");
   533→      return;
   534→    }
   535→    if (checking || result || !word) return;
   536→    setChecking(true);
   537→    try {
   538→      stop();
   539→      const res = await checkWord({
   540→        wordId: word.id,
   541→        answer: val,
   542→        mode: "DICTATION",
   543→        durationMs: Date.now() - startRef.current,
   544→      });
   545→      setResult(res);
   546→      attemptsRef.current = [
   547→        ...attemptsRef.current,
   548→        { word: res.word ?? word, correct: res.correct, result: res },
   549→      ];
   550→    } catch {
   551→      toast.error("提交失败，请再试一次");
   552→    } finally {
   553→      setChecking(false);
   554→    }
   555→  }, [answer, checking, result, word, stop]);
   556→
   557→  const next = useCallback(() => {
   558→    if (index + 1 >= words.length) {
   559→      onFinish(attemptsRef.current);
   560→      return;
   561→    }
   562→    setResult(null);
   563→    setAnswer("");
   564→    setIndex((i) => i + 1);
   565→  }, [index, words.length, onFinish]);
   566→
   567→  /* 答对 + 自动下一个：1.5s 后前进 */
   568→  useEffect(() => {
   569→    if (result?.correct && autoNext) {
   570→      const t = setTimeout(next, 1500);
   571→      return () => clearTimeout(t);
   572→    }
   573→  }, [result, autoNext, next]);
   574→
   575→  if (!word) return null;
   576→
   577→  return (
   578→    <div className="flex flex-col gap-5">
   579→      {/* 顶部进度 */}
   580→      <div className="flex items-center gap-3">
   581→        <Button variant="ghost" size="icon" className="size-11 shrink-0 rounded-full" onClick={onQuit} aria-label="退出听写">
   582→          <ArrowLeft className="size-5" aria-hidden />
   583→        </Button>
   584→        <Progress value={progress} className="h-2.5 flex-1" aria-label={`第 ${index + 1} 个，共 ${words.length} 个`} />
   585→        <span className="shrink-0 text-sm font-bold text-muted-foreground">
   586→          第 {index + 1} / {words.length} 词
   587→        </span>
   588→      </div>
   589→
   590→      <motion.div key={word.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ duration: 0.2 }}>
   591→        <Card className="rounded-2xl">
   592→          <CardContent className="flex flex-col items-center gap-5 p-6">
   593→            {/* 大播放按钮 */}
   594→            <motion.button
   595→              type="button"
   596→              whileTap={{ scale: 0.92 }}
   597→              onClick={() => void speak(word.headword, { speed })}
   598→              aria-label={`重播 ${word.headword} 的发音`}
   599→              className={cn(
   600→                "flex size-20 items-center justify-center rounded-full bg-primary text-primary-foreground shadow-lg transition-shadow hover:shadow-xl",
   601→                speaking && "ring-4 ring-primary/30 ring-offset-2"
   602→              )}
   603→            >
   604→              <Volume2 className="size-9" aria-hidden />
   605→            </motion.button>
   606→            <p className="-mt-2 text-xs text-muted-foreground">点喇叭可以再听一遍</p>
   607→
   608→            {/* 提示区 */}
   609→            <div className="flex min-h-11 w-full items-center justify-center rounded-xl bg-secondary/50 px-4 py-2.5">
   610→              <HintArea level={hintLevel} word={word} />
   611→            </div>
   612→
   613→            {/* 输入 + 提交 */}
   614→            {!result ? (
   615→              <>
   616→                <Input
   617→                  ref={inputRef}
   618→                  value={answer}
   619→                  onChange={(e) => setAnswer(e.target.value)}
   620→                  onKeyDown={(e) => {
   621→                    if (e.key === "Enter") {
   622→                      e.preventDefault();
   623→                      void submit();
   624→                    }
   625→                  }}
   626→                  disabled={checking}
   627→                  placeholder="输入你听到的单词"
   628→                  className="h-14 rounded-2xl text-center font-mono text-2xl font-bold"
   629→                  autoComplete="off"
   630→                  autoCapitalize="off"
   631→                  autoCorrect="off"
   632→                  spellCheck={false}
   633→                  aria-label="拼写输入框"
   634→                />
   635→                <Button
   636→                  className="h-13 w-full rounded-2xl py-4 text-base font-bold"
   637→                  onClick={() => void submit()}
   638→                  disabled={checking}
   639→                >
   640→                  {checking ? (
   641→                    <>
   642→                      <Loader2 className="size-5 animate-spin" aria-hidden />
   643→                      正在批改…
   644→                    </>
   645→                  ) : (
   646→                    <>
   647→                      <Check className="size-5" aria-hidden />
   648→                      提交（按回车也行）
   649→                    </>
   650→                  )}
   651→                </Button>
   652→              </>
   653→            ) : (
   654→              <>
   655→                {/* 批改反馈 */}
   656→                <motion.div
   657→                  initial={{ opacity: 0, y: 8 }}
   658→                  animate={{ opacity: 1, y: 0 }}
   659→                  className="w-full rounded-2xl border border-border/70 bg-card p-4"
   660→                >
   661→                  <CheckFeedback result={result} />
   662→                </motion.div>
   663→                <Button
   664→                  className="h-13 w-full rounded-2xl py-4 text-base font-bold"
   665→                  onClick={next}
   666→                  aria-label="下一个词"
   667→                >
   668→                  {index + 1 >= words.length ? "看结果" : "下一词"}
   669→                </Button>
   670→              </>
   671→            )}
   672→          </CardContent>
   673→        </Card>
   674→      </motion.div>
   675→    </div>
   676→  );
   677→}
   678→
   679→/* ============================================================
   680→ * 阶段三：result 结果页
   681→ * ============================================================ */
   682→
   683→function ResultStage({
   684→  attempts,
   685→  onRetryWrong,
   686→  onChangeList,
   687→  onHome,
   688→}: {
   689→  attempts: Attempt[];
   690→  onRetryWrong: () => void;
   691→  onChangeList: () => void;
   692→  onHome: () => void;
   693→}) {
   694→  const { speak } = useSpeech();
   695→  const total = attempts.length;
   696→  const correctCount = attempts.filter((a) => a.correct).length;
   697→  const accuracy = total ? Math.round((correctCount / total) * 100) : 0;
   698→  const wrong = attempts.filter((a) => !a.correct);
   699→
   700→  useEffect(() => {
   701→    toast.success(`听写完成！答对 ${correctCount} / ${total}`, {
   702→      description: accuracy >= 80 ? "太厉害了，继续保持！" : "错词可以再来一轮哦～",
   703→    });
   704→    // 仅在进入结果页时提示一次
   705→  }, [correctCount, total, accuracy]);
   706→
   707→  const ringColor = accuracy >= 80 ? "var(--chart-3)" : accuracy >= 60 ? "var(--chart-2)" : "var(--chart-4)";
   708→
   709→  return (
   710→    <div className="flex flex-col gap-5">
   711→      <motion.div initial={{ scale: 0.92, opacity: 0 }} animate={{ scale: 1, opacity: 1 }} className="flex flex-col items-center gap-3 pt-2 text-center">
   712→        <h1 className="text-2xl font-extrabold tracking-tight">听写结果</h1>
   713→        <AccuracyRing percent={accuracy} color={ringColor} subLabel={`答对 ${correctCount} / ${total} 词`} />
   714→      </motion.div>
   715→
   716→      {wrong.length > 0 ? (
   717→        <Card className="rounded-2xl">
   718→          <CardContent className="p-4">
   719→            <p className="mb-3 font-bold">错词回顾（点喇叭重听）</p>
   720→            <div className="flex max-h-72 flex-col gap-2 overflow-y-auto nice-scrollbar pr-1">
   721→              {wrong.map((a, i) => (
   722→                <div key={`${a.word.id}-${i}`} className="rounded-xl border border-border/70 p-3">
   723→                  <div className="flex items-center justify-between gap-2">
   724→                    <div className="min-w-0">
   725→                      <p className="truncate font-mono font-bold">
   726→                        {a.word.headword}
   727→                        <span className="ml-2 text-xs font-normal text-muted-foreground">/{a.word.ipa}/</span>
   728→                      </p>
   729→                      <p className="truncate text-xs text-muted-foreground">{a.word.translation}</p>
   730→                    </div>
   731→                    <Button
   732→                      variant="outline"
   733→                      size="icon"
   734→                      className="size-10 shrink-0 rounded-full"
   735→                      onClick={() => void speak(a.word.headword)}
   736→                      aria-label={`重听 ${a.word.headword}`}
   737→                    >
   738→                      <Volume2 className="size-4" aria-hidden />
   739→                    </Button>
   740→                  </div>
   741→                  <Badge variant="secondary" className="mt-2 rounded-full bg-rose-100 text-[11px] text-rose-700">
   742→                    错因：{(a.result.errorTypes ?? []).length ? a.result.errorTypes.join("、") : "待分析"}
   743→                  </Badge>
   744→                </div>
   745→              ))}
   746→            </div>
   747→          </CardContent>
   748→        </Card>
   749→      ) : (
   750→        <Card className="rounded-2xl border-emerald-200 bg-emerald-50/60">
   751→          <CardContent className="p-5 text-center text-sm font-bold text-emerald-700">
   752→            全部答对，一个错词都没有，太棒了！
   753→          </CardContent>
   754→        </Card>
   755→      )}
   756→
   757→      <div className="flex flex-col gap-3 sm:flex-row">
   758→        {wrong.length > 0 ? (
   759→          <Button className="h-12 flex-1 rounded-2xl text-base font-bold" onClick={onRetryWrong}>
   760→            <RefreshCw className="size-4" aria-hidden />
   761→            重听错词（{wrong.length}）
   762→          </Button>
   763→        ) : null}
   764→        <Button variant="outline" className="h-12 flex-1 rounded-2xl text-base font-bold" onClick={onChangeList}>
   765→          换个词单
   766→        </Button>
   767→        <Button variant="outline" className="h-12 flex-1 rounded-2xl text-base font-bold" onClick={onHome}>
   768→          <Home className="size-4" aria-hidden />
   769→          回首页
   770→        </Button>
   771→      </div>
   772→    </div>
   773→  );
   774→}
